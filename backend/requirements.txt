# --- Speaker diarization (pyannote.audio on CPU) ----------------------------
# Around 1.5 GB of wheels. Only the diarization step needs them, and the app
# runs without them (turn "Identify speakers" off on the Settings page), but
# `npm run setup` installs everything so the default experience is complete.
# These entries are also kept standalone in requirements-ml.txt.
torch==2.5.1+cpu
torchaudio==2.5.1+cpu
pyannote.audio==3.3.2

# --- pins that are load-bearing, not cosmetic -------------------------------
# huggingface_hub 1.x removed the `use_auth_token` kwarg that pyannote 3.3.2
# still passes to hf_hub_download, so an unpinned install fails at
# Pipeline.from_pretrained with "unexpected keyword argument 'use_auth_token'".
# Stay on 0.26.x until pyannote.audio 4.x.
huggingface_hub==0.26.5
# pyannote's stack (asteroid-filterbanks, pyannote.core) is still numpy 1.x.
numpy==1.26.4
# torchaudio 2.5 has no built-in Windows audio backend; soundfile (libsndfile)
# is what actually decodes the wav. Without it loading audio raises on Windows.
soundfile==0.12.1

# Transitive but pinned so a future release cannot silently break the pipeline.
lightning==2.6.5
pytorch-lightning==2.6.5
speechbrain==1.1.1
